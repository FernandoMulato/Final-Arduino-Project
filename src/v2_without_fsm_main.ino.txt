constexpr uint8_t LCD_EN = 11;
constexpr uint8_t LCD_D4 = 5;
constexpr uint8_t LCD_D5 = 4;
constexpr uint8_t LCD_D6 = 3;
constexpr uint8_t LCD_D7 = 2;

// --- Pines Keypad ---
byte ROW_PINS[4] = {29, 31, 33, 35};
byte COL_PINS[4] = {37, 39, 41, 43};

// --- Pines LED RGB ---
constexpr uint8_t PIN_LED_R = 22;
constexpr uint8_t PIN_LED_G = 24;
constexpr uint8_t PIN_LED_B = 26;

// --- Pines actuadores ---
constexpr uint8_t PIN_SERVO  = 10;
constexpr uint8_t PIN_BUZZER = 9;

// --- Pines sensores ---
constexpr uint8_t PIN_MIC  = A0;
constexpr uint8_t PIN_NTC  = A1;
constexpr uint8_t PIN_LDR  = A2;
constexpr uint8_t PIN_DOOR = 21;

// --- Servo ---
constexpr uint8_t SRV_LOCKED   = 0;
constexpr uint8_t SRV_UNLOCKED = 90;

// --- Umbrales sensores ---
constexpr float TEMP_LOW   = 20.0f;
constexpr float TEMP_HIGH  = 50.0f;
constexpr int   LIGHT_MIN  = 100;
constexpr int   SOUND_HIGH = 800;

// --- NTC Steinhart-Hart ---
constexpr float R1 = 10000.0f;
constexpr float C1 = 0.001129148f;
constexpr float C2 = 0.000234125f;
constexpr float C3 = 0.0000000876741f;

// --- EEPROM ---
constexpr uint8_t EEP_MAGIC       = 0;
constexpr uint8_t EEP_USER_COUNT  = 1;
constexpr uint8_t EEP_USERS_START = 2;
constexpr uint8_t EEP_USER_SIZE   = 24;
constexpr uint8_t EEP_MAGIC_VAL   = 0xA5;
constexpr uint8_t MAX_USERS       = 10;
constexpr uint8_t PIN_MAX_USES    = 4;
constexpr uint8_t PIN_HIST_LEN    = 4;
constexpr uint8_t PIN_MIN_LEN     = 4;
constexpr uint8_t PIN_MAX_LEN     = 6;
constexpr uint8_t OFF_PIN         = 0;
constexpr uint8_t OFF_ROLE        = 4;
constexpr uint8_t OFF_USES        = 5;
constexpr uint8_t OFF_ACTIVE      = 6;
constexpr uint8_t OFF_HIST_IDX    = 7;
constexpr uint8_t OFF_HIST        = 8;

constexpr const char* ROLE_NAMES[5] = {
  "", "Security", "Operator", "Coord", "Manager"
};

// --- Keypad ---
char keyMap[4][4] = {
  {'1','2','3','A'}, {'4','5','6','B'},
  {'7','8','9','C'}, {'*','0','#','D'}
};
Keypad keypad = Keypad(makeKeymap(keyMap), ROW_PINS, COL_PINS, 4, 4);

// --- LCD ---
LiquidCrystal lcd(LCD_RS, LCD_EN, LCD_D4, LCD_D5, LCD_D6, LCD_D7);

// --- Servo ---
Servo doorServo;

// ============================================================================
// FSM
// ============================================================================
enum State : uint8_t {
  S_IDLE, S_OPEN, S_BLOCKED,
  S_ENV_MONITOR, S_INTRUSION_MONITOR, S_ALARM
};

State currentState = S_IDLE;
State nextState    = S_IDLE;
bool  stateChanged = false;

void changeState(State s) {
  nextState    = s;
  stateChanged = true;
}

// --- PIN entry ---
char    pinBuf[7] = {0};
uint8_t pinLen    = 0;
uint8_t failCount = 0;

// --- Usuario autenticado ---
uint8_t lastAuthRole    = 0;
uint8_t lastAuthUserIdx = 0xFF;

// --- PIN change ---
bool    pinChangeRequired = false;
uint8_t pinChangeUserIdx  = 0xFF;

// --- Menu ---
bool    menuActive  = false;
uint8_t menuStep    = 0;
char    menuBuf[7]  = {0};
uint8_t menuBufLen  = 0;
uint8_t menuUserIdx = 0xFF;

// --- Timers ---
unsigned long stateEntryTime  = 0;
unsigned long lastLcdUpdate   = 0;
unsigned long lastSensorRead  = 0;
constexpr unsigned long T_UNLOCK        = 2000;
constexpr unsigned long T_LOCKOUT       = 5000;
constexpr unsigned long T_ALARM         = 2000;
constexpr unsigned long T_ENV_MONITOR   = 4000;
constexpr unsigned long T_INTRUSION     = 2000;
constexpr unsigned long T_TRIPLE        = 12000;
constexpr unsigned long LCD_INTERVAL    = 250;
constexpr unsigned long SENSOR_INTERVAL = 200;

// --- Blink ---
unsigned long lastBlink   = 0;
bool          ledOn       = false;
bool          blinkActive = false;
unsigned long blinkOnMs   = 0;
unsigned long blinkOffMs  = 0;

// --- Sensores ---
float temperature = 25.0f;
int   lightLevel  = 500;
int   micVal      = 0;
bool  doorOpen    = false;

// --- Interrupcion puerta ---
volatile bool doorChanged = false;

// --- Triple alarma ---
uint8_t       alarmCount     = 0;
unsigned long firstAlarmTime = 0;

// ============================================================================
// ISR puerta
// ============================================================================
void doorISR() {
  doorChanged = true;
}

// ============================================================================
// EEPROM helpers
// ============================================================================
void loadUser(uint8_t idx, char pin[5], uint8_t& role, uint8_t& uses,
              bool& active, uint8_t& histIdx, char history[4][4]) {
  uint16_t addr = EEP_USERS_START + idx * EEP_USER_SIZE;
  for (uint8_t i = 0; i < 4; i++) pin[i] = EEPROM.read(addr + OFF_PIN + i);
  pin[4]  = '\0';
  role    = EEPROM.read(addr + OFF_ROLE);
  uses    = EEPROM.read(addr + OFF_USES);
  active  = EEPROM.read(addr + OFF_ACTIVE) != 0;
  histIdx = EEPROM.read(addr + OFF_HIST_IDX);
  for (uint8_t h = 0; h < PIN_HIST_LEN; h++)
    for (uint8_t i = 0; i < 4; i++)
      history[h][i] = EEPROM.read(addr + OFF_HIST + h * 4 + i);
  if (pin[0] < '0' || pin[0] > '9') active = false;
}

void saveUser(uint8_t idx, const char pin[5], uint8_t role, uint8_t uses,
              bool active, uint8_t histIdx, const char history[4][4]) {
  uint16_t addr = EEP_USERS_START + idx * EEP_USER_SIZE;
  for (uint8_t i = 0; i < 4; i++) EEPROM.update(addr + OFF_PIN + i, pin[i]);
  EEPROM.update(addr + OFF_ROLE,     role);
  EEPROM.update(addr + OFF_USES,     uses);
  EEPROM.update(addr + OFF_ACTIVE,   active ? 1 : 0);
  EEPROM.update(addr + OFF_HIST_IDX, histIdx);
  for (uint8_t h = 0; h < PIN_HIST_LEN; h++)
    for (uint8_t i = 0; i < 4; i++)
      EEPROM.update(addr + OFF_HIST + h * 4 + i, history[h][i]);
}

uint8_t userCount() { return EEPROM.read(EEP_USER_COUNT); }

uint8_t findUserByPin(const char* pin) {
  uint8_t n = userCount();
  for (uint8_t i = 0; i < n; i++) {
    char sp[5]; uint8_t r, u, hi; bool a; char hist[4][4];
    loadUser(i, sp, r, u, a, hi, hist);
    if (!a) continue;
    bool match = true;
    for (uint8_t j = 0; j < 4; j++)
      if (pin[j] != sp[j]) { match = false; break; }
    if (match) return i;
  }
  return 0xFF;
}

void addUser(const char* pin, uint8_t role) {
  uint8_t n = userCount();
  if (n >= MAX_USERS) return;
  char hist[4][4]; memset(hist, 0, sizeof(hist));
  char p[5];
  for (uint8_t i = 0; i < 4; i++) p[i] = pin[i];
  p[4] = '\0';
  saveUser(n, p, role, 0, true, 0, (const char(*)[4])hist);
  EEPROM.update(EEP_USER_COUNT, n + 1);
  Serial.print(F("[EEPROM] Agregado: PIN="));
  Serial.print(pin);
  Serial.print(F(" Rol="));
  Serial.println(ROLE_NAMES[role]);
}

// ============================================================================
// PIN helpers
// ============================================================================
bool pinIsUnique(uint8_t userIdx, const char* newPin) {
  char sp[5]; uint8_t r, u, hi; bool a; char hist[4][4];
  loadUser(userIdx, sp, r, u, a, hi, hist);
  bool same = true;
  for (uint8_t j = 0; j < 4; j++)
    if (newPin[j] != sp[j]) { same = false; break; }
  if (same) return false;
  for (uint8_t h = 0; h < PIN_HIST_LEN; h++) {
    if (hist[h][0] < '0' || hist[h][0] > '9') continue;
    same = true;
    for (uint8_t j = 0; j < 4; j++)
      if (newPin[j] != hist[h][j]) { same = false; break; }
    if (same) return false;
  }
  return true;
}

void rotatePin(uint8_t userIdx, const char* newPin) {
  char sp[5]; uint8_t r, u, hi; bool a; char hist[4][4];
  loadUser(userIdx, sp, r, u, a, hi, hist);
  for (uint8_t i = 0; i < 4; i++) hist[hi][i] = sp[i];
  hi = (hi + 1) % PIN_HIST_LEN;
  saveUser(userIdx, newPin, r, 0, true, hi, (const char(*)[4])hist);
  Serial.println(F("[PIN] Rotacion OK, usos=0"));
}

// ============================================================================
// Autenticacion
// ============================================================================
bool validateAccess(const char* pin) {
  uint8_t idx = findUserByPin(pin);
  if (idx == 0xFF) { Serial.println(F("[AUTH] PIN no encontrado")); return false; }
  char sp[5]; uint8_t role, uses, hi; bool a; char hist[4][4];
  loadUser(idx, sp, role, uses, a, hi, hist);
  if (role < 1 || role > 4) { Serial.println(F("[AUTH] Rol invalido")); return false; }
  uses++;
  lastAuthRole    = role;
  lastAuthUserIdx = idx;
  Serial.print(F("[AUTH] User ")); Serial.print(idx);
  Serial.print(F(" usos=")); Serial.print(uses);
  Serial.print(F("/")); Serial.println(PIN_MAX_USES);
  if (uses >= PIN_MAX_USES) {
    pinChangeRequired = true;
    pinChangeUserIdx  = idx;
    Serial.println(F("[AUTH] *** PIN EXPIRADO ***"));
  }
  saveUser(idx, sp, role, uses, true, hi, (const char(*)[4])hist);
  Serial.print(F("[AUTH] OK — ")); Serial.println(ROLE_NAMES[role]);
  return true;
}

// ============================================================================
// Actuadores
// ============================================================================
void setRGB(bool r, bool g, bool b) {
  digitalWrite(PIN_LED_R, r ? HIGH : LOW);
  digitalWrite(PIN_LED_G, g ? HIGH : LOW);
  digitalWrite(PIN_LED_B, b ? HIGH : LOW);
}
void ledOff()   { setRGB(false,false,false); blinkActive=false; }
void ledRed()   { setRGB(true, false,false); blinkActive=false; }
void ledGreen() { setRGB(false,true, false); blinkActive=false; }
void ledBlue()  { setRGB(false,false,true);  blinkActive=false; }

void startBlink(unsigned long onMs, unsigned long offMs) {
  blinkOnMs=onMs; blinkOffMs=offMs;
  blinkActive=true; lastBlink=millis(); ledOn=false;
}
void updateBlink() {
  if (!blinkActive) return;
  unsigned long now=millis();
  unsigned long interval = ledOn ? blinkOffMs : blinkOnMs;
  if (now-lastBlink >= interval) {
    lastBlink=now; ledOn=!ledOn;
    digitalWrite(PIN_LED_R, ledOn ? HIGH : LOW);
  }
}
void setBuzzer(bool on) { if(on) tone(PIN_BUZZER,1000); else noTone(PIN_BUZZER); }
void unlockDoor() { doorServo.write(SRV_UNLOCKED); }
void lockDoor()   { doorServo.write(SRV_LOCKED);   }

// ============================================================================
// Sensores
// ============================================================================
void readSensors() {
  micVal     = analogRead(PIN_MIC);
  lightLevel = analogRead(PIN_LDR);

  // Puerta via ISR
  if (doorChanged) {
    doorChanged = false;
    doorOpen    = digitalRead(PIN_DOOR) == LOW;
  }

  // Temperatura NTC
  int ntcRaw = analogRead(PIN_NTC);
  if (ntcRaw > 0) {
    float R2    = R1 * (1023.0f / ntcRaw - 1.0f);
    float logR2 = log(R2);
    temperature = (1.0f/(C1+C2*logR2+C3*logR2*logR2*logR2))-273.15f;
  }
}

// ============================================================================
// LCD
// ============================================================================
void updateDisplay() {
  lcd.clear();
  lcd.setCursor(0,0);

  if (menuActive) {
    lcd.print(F("CAMBIO DE PIN"));
    lcd.setCursor(0,1);
    if (menuStep == 0) {
      lcd.print(F("Actual:"));
    } else {
      lcd.print(F("Nuevo:"));
    }
    for (uint8_t i=0; i<menuBufLen; i++) lcd.print('*');
    return;
  }

  switch (currentState) {
    case S_IDLE:
      lcd.print(F("IDLE"));
      lcd.setCursor(0,1);
      if (pinLen > 0) {
        for (uint8_t i=0; i<6; i++) lcd.print(i<pinLen ? '*' : '-');
        lcd.print(F(" #=ok"));
      } else {
        lcd.print(F("Enter PIN: *=menu"));
      }
      break;

    case S_OPEN:
      lcd.print(F("ACCESS GRANTED"));
      lcd.setCursor(0,1);
      lcd.print(ROLE_NAMES[lastAuthRole]);
      lcd.print(F(" "));
      lcd.print((T_UNLOCK-(millis()-stateEntryTime))/1000);
      lcd.print('s');
      break;

    case S_BLOCKED:
      lcd.print(F("BLOCKED"));
      lcd.setCursor(0,1);
      lcd.print(F("Wait 5s..."));
      break;

    case S_ENV_MONITOR:
      lcd.print(F("MONITOR ENV"));
      lcd.setCursor(0,1);
      lcd.print(F("T:")); lcd.print((int)temperature);
      lcd.print(F("C L:")); lcd.print(lightLevel);
      break;

    case S_INTRUSION_MONITOR:
      lcd.print(F("MONITOR SEC"));
      lcd.setCursor(0,1);
      lcd.print(doorOpen ? F("PUERTA: ABIERTA") : F("PUERTA: CERRADA"));
      break;

    case S_ALARM:
      lcd.print(F("!!! ALARM !!!"));
      lcd.setCursor(0,1);
      lcd.print(F("Intrusion! #"));
      lcd.print(alarmCount);
      break;
  }
}

// ============================================================================
// Menu cambio PIN
// ============================================================================
void openMenu() {
  menuActive=true; menuStep=0; menuBufLen=0;
  menuUserIdx=0xFF; memset(menuBuf,0,sizeof(menuBuf));
  Serial.println(F("[MENU] Ingrese PIN actual:"));
}

void closeMenu() {
  menuActive=false; menuStep=0; menuBufLen=0;
  menuUserIdx=0xFF; memset(menuBuf,0,sizeof(menuBuf));
  Serial.println(F("[MENU] Cerrado"));
}

void handleMenuKey(char key) {
  if (key == '*') { closeMenu(); return; }

  if (key >= '0' && key <= '9') {
    if (menuBufLen < PIN_MAX_LEN) menuBuf[menuBufLen++] = key;
    return;
  }

  if (key == '#') {
    menuBuf[menuBufLen] = '\0';
    if (menuStep == 0) {
      if (menuBufLen < PIN_MIN_LEN) {
        Serial.println(F("[MENU] PIN muy corto"));
        menuBufLen=0; memset(menuBuf,0,sizeof(menuBuf)); return;
      }
      uint8_t idx = findUserByPin(menuBuf);
      if (idx == 0xFF) {
        Serial.println(F("[MENU] PIN incorrecto"));
        menuBufLen=0; memset(menuBuf,0,sizeof(menuBuf)); return;
      }
      menuUserIdx=idx; menuStep=1;
      menuBufLen=0; memset(menuBuf,0,sizeof(menuBuf));
      Serial.println(F("[MENU] OK. PIN nuevo (min 4):"));
    } else {
      if (menuBufLen < PIN_MIN_LEN) {
        Serial.println(F("[MENU] PIN muy corto"));
        menuBufLen=0; memset(menuBuf,0,sizeof(menuBuf)); return;
      }
      if (!pinIsUnique(menuUserIdx, menuBuf)) {
        Serial.println(F("[MENU] PIN ya usado. Elija otro."));
        menuBufLen=0; memset(menuBuf,0,sizeof(menuBuf)); return;
      }
      rotatePin(menuUserIdx, menuBuf);
      if (pinChangeRequired && pinChangeUserIdx == menuUserIdx) {
        pinChangeRequired=false; pinChangeUserIdx=0xFF;
      }
      Serial.println(F("[MENU] *** PIN cambiado OK ***"));
      lcd.clear();
      lcd.setCursor(0,0); lcd.print(F("PIN cambiado!"));
      lcd.setCursor(0,1); lcd.print(F("OK"));
      delay(1500);
      closeMenu();
    }
  }
}

// ============================================================================
// Keypad principal
// ============================================================================
void handleKey(char key) {
  if (key >= '0' && key <= '9') {
    if (pinLen < 6) {
      pinBuf[pinLen++] = key;
      Serial.print(F("[KEY] ")); Serial.println(key);
    }
  } else if (key == '#') {
    if (pinLen >= PIN_MIN_LEN) {
      pinBuf[pinLen] = '\0';
      // Verificar PIN expirado
      if (pinChangeRequired) {
        uint8_t idx = findUserByPin(pinBuf);
        if (idx != 0xFF && idx == pinChangeUserIdx) {
          Serial.println(F("[AUTH] PIN expirado. Presione * para cambiar."));
          lcd.clear();
          lcd.setCursor(0,0); lcd.print(F("PIN EXPIRADO"));
          lcd.setCursor(0,1); lcd.print(F("Presione *"));
          delay(2000);
          pinLen=0; memset(pinBuf,0,sizeof(pinBuf));
          return;
        }
      }
      if (validateAccess(pinBuf)) {
        changeState(S_OPEN);
      } else {
        failCount++;
        Serial.print(F("[AUTH] FAIL #")); Serial.println(failCount);
        if (failCount >= 3) changeState(S_BLOCKED);
        else { pinLen=0; memset(pinBuf,0,sizeof(pinBuf)); }
      }
    } else {
      Serial.println(F("[KEY] PIN muy corto"));
    }
  } else if (key == '*') {
    if (pinLen > 0) {
      pinLen=0; memset(pinBuf,0,sizeof(pinBuf));
      Serial.println(F("[KEY] Buffer limpiado"));
    } else {
      openMenu();
    }
  }
}

// ============================================================================
// Entradas a estados
// ============================================================================
void enterState(State s) {
  stateEntryTime=millis(); lastLcdUpdate=0; currentState=s;
  switch (s) {
    case S_IDLE:
      failCount=0; pinLen=0; memset(pinBuf,0,sizeof(pinBuf));
      menuActive=false;
      ledOff(); setBuzzer(false); lockDoor();
      Serial.println(F("[STATE] IDLE"));
      break;
    case S_OPEN:
      ledGreen(); setBuzzer(false); unlockDoor();
      Serial.println(F("[STATE] OPEN — 2s"));
      break;
    case S_BLOCKED:
      startBlink(300,700); setBuzzer(false); lockDoor();
      Serial.println(F("[STATE] BLOCKED — 5s"));
      break;
    case S_ENV_MONITOR:
      ledBlue(); setBuzzer(false); alarmCount=0;
      Serial.println(F("[STATE] ENV_MONITOR — 4s"));
      break;
    case S_INTRUSION_MONITOR:
      ledBlue(); setBuzzer(false); alarmCount=0;
      doorOpen=false; doorChanged=false;
      Serial.println(F("[STATE] INTRUSION_MONITOR — 2s"));
      break;
    case S_ALARM:
      startBlink(100,500); setBuzzer(true);
      alarmCount++;
      if (alarmCount==1) firstAlarmTime=millis();
      Serial.print(F("[STATE] ALARM #")); Serial.println(alarmCount);
      break;
  }
}

// ============================================================================
// Logica de estados
// ============================================================================
void updateState() {
  unsigned long now     = millis();
  unsigned long elapsed = now - stateEntryTime;

  if (now-lastSensorRead >= SENSOR_INTERVAL) {
    lastSensorRead=now;
    readSensors();
  }

  switch (currentState) {
    case S_OPEN:
      if (elapsed >= T_UNLOCK) {
        pinLen=0; memset(pinBuf,0,sizeof(pinBuf));
        changeState(S_IDLE);
      }
      break;

    case S_BLOCKED:
      if (elapsed >= T_LOCKOUT) changeState(S_IDLE);
      break;

    case S_ENV_MONITOR:
      if (temperature < TEMP_LOW || temperature > TEMP_HIGH ||
          lightLevel < LIGHT_MIN) {
        Serial.print(F("[ENV] Umbral! T="));
        Serial.print(temperature);
        Serial.print(F(" L=")); Serial.println(lightLevel);
        changeState(S_ALARM);
      } else if (elapsed >= T_ENV_MONITOR) {
        Serial.println(F("[ENV] Sin eventos -> IDLE"));
        changeState(S_IDLE);
      }
      break;

    case S_INTRUSION_MONITOR:
      if (doorOpen) {
        Serial.println(F("[INTRUSION] Puerta abierta!"));
        changeState(S_ALARM);
      } else if (micVal > SOUND_HIGH) {
        Serial.print(F("[INTRUSION] Sonido: ")); Serial.println(micVal);
        changeState(S_ALARM);
      } else if (elapsed >= T_INTRUSION) {
        Serial.println(F("[INTRUSION] Sin eventos -> IDLE"));
        changeState(S_IDLE);
      }
      break;

    case S_ALARM:
      if (alarmCount >= 3 && (now-firstAlarmTime) < T_TRIPLE) {
        Serial.println(F("[ALARM] Triple! Bloqueo extendido"));
        alarmCount=0; changeState(S_BLOCKED);
      } else if (elapsed >= T_ALARM) {
        changeState(S_INTRUSION_MONITOR);
      }
      break;

    case S_IDLE:
      break;
  }

  updateBlink();

  if (now-lastLcdUpdate >= LCD_INTERVAL) {
    lastLcdUpdate=now;
    updateDisplay();
  }
}

void applyStateChange() {
  if (stateChanged) { stateChanged=false; enterState(nextState); }
}

// ============================================================================
// Setup
// ============================================================================
void setup() {
  Serial.begin(9600);

  pinMode(PIN_LED_R, OUTPUT);
  pinMode(PIN_LED_G, OUTPUT);
  pinMode(PIN_LED_B, OUTPUT);
  ledOff();

  pinMode(PIN_BUZZER, OUTPUT);
  setBuzzer(false);

  pinMode(PIN_DOOR, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PIN_DOOR), doorISR, CHANGE);

  doorServo.attach(PIN_SERVO);
  lockDoor();

  lcd.begin(16, 2);
  lcd.print(F("Access Control"));
  lcd.setCursor(0,1);
  lcd.print(F("Starting..."));

  if (EEPROM.read(EEP_MAGIC) != EEP_MAGIC_VAL) {
    EEPROM.write(EEP_MAGIC,      EEP_MAGIC_VAL);
    EEPROM.write(EEP_USER_COUNT, 0);
    addUser("1234", 4);  // Manager
    addUser("5678", 2);  // Operator
    addUser("9999", 1);  // Security
    Serial.println(F("[EEPROM] Usuarios de prueba cargados"));
  } else {
    Serial.println(F("[EEPROM] Usuarios existentes"));
  }

  uint8_t n = userCount();
  Serial.print(F("[EEPROM] Total usuarios: ")); Serial.println(n);
  for (uint8_t i=0; i<n; i++) {
    char sp[5]; uint8_t r,u,hi; bool a; char hist[4][4];
    loadUser(i, sp, r, u, a, hi, hist);
    Serial.print(F("  [")); Serial.print(i);
    Serial.print(F("] PIN=")); Serial.print(sp);
    Serial.print(F(" Rol=")); Serial.print(ROLE_NAMES[r]);
    Serial.print(F(" Usos=")); Serial.print(u);
    Serial.print(F(" Activo=")); Serial.println(a ? F("SI") : F("NO"));
  }

  delay(1000);
  enterState(S_IDLE);

  Serial.println(F("=== SISTEMA FINAL ==="));
  Serial.println(F("PINs: 1234=Manager | 5678=Operator | 9999=Security"));
  Serial.println(F("A=ENV_MONITOR | B=INTRUSION_MONITOR"));
  Serial.println(F("* sin digitos=menu | * con digitos=limpiar"));
}

// ============================================================================
// Loop
// ============================================================================
void loop() {
  char key = keypad.getKey();

  if (currentState == S_IDLE) {
    if (menuActive) {
      if (key != NO_KEY) handleMenuKey(key);
    } else {
      if      (key == 'A') { Serial.println(F("[TEST] ENV_MONITOR"));       changeState(S_ENV_MONITOR);       }
      else if (key == 'B') { Serial.println(F("[TEST] INTRUSION_MONITOR")); changeState(S_INTRUSION_MONITOR); }
      else if (key != NO_KEY) handleKey(key);
    }
  }

  updateState();
  applyStateChange();
}
